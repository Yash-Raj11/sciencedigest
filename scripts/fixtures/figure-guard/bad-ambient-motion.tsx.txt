import { useEffect, useState } from "react";
import { Figure } from "@/components/figures/Figure";

// A dot that pulses for ever: motion no fact drives.
export default function Fixture({ alt }: { alt: string }) {
  const [on, setOn] = useState(false);
  useEffect(() => {
    const id = setInterval(() => setOn((v) => !v), Number(alt.length));
    return () => clearInterval(id);
  }, [alt]);
  return (
    <Figure title="A fixture" desc={alt} height={120}>
      <circle cx={20} cy={20} r={on ? 8 : 6}>
        <animate attributeName="opacity" dur="s" repeatCount="indefinite" />
      </circle>
    </Figure>
  );
}
